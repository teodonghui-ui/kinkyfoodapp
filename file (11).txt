                         <div className="w-16 h-16 bg-[#FF2A6D]/20 border border-[#FF2A6D] rounded-full flex items-center justify-center mb-6 shadow-glow text-[#FF2A6D]"><Check size={32} /></div>
                         <h3 className="text-xl font-black italic text-[#FCE4EC] mb-2 uppercase">Ritual Complete</h3>
                         <p className="text-[#AD6688] text-xs italic opacity-70">Sent into the abyss. Wait for his decree. 💋</p>
                      </div>
                    ) : (
                      <div className="flex-1 overflow-y-auto custom-scrollbar flex flex-col">
                        <div className="p-4 space-y-4 flex-grow">
                          {cart.length === 0 ? (
                            <div className="h-40 flex flex-col items-center justify-center opacity-20 text-[#FF2A6D]"><ShoppingBag size={40} /><p className="mt-4 font-black italic text-sm">The tray is empty...</p></div>
                          ) : (
                            cart.map(item => (
                              <div key={item.cartId} className="flex gap-4 items-center bg-[#0A0206]/80 p-4 rounded-2xl border border-[#FF2A6D]/10 relative overflow-hidden transition-all hover:bg-[#1A030C]">
                                <div className="flex-1 min-w-0 z-10">
                                  <p className="font-black text-sm text-[#FCE4EC] italic leading-tight mb-1 truncate uppercase">{item.name}</p>
                                  <p className="text-[#FF2A6D] font-black text-[9px] uppercase tracking-widest">Sacrifice: ${(item.price * item.quantity).toFixed(2)}</p>
                                </div>
                                <div className="flex items-center gap-3 bg-[#16040C] p-1.5 rounded-xl border border-[#FF2A6D]/10 z-10 flex-shrink-0">
                                  <button onClick={() => { if (item.quantity === 1) setCart(prev => prev.filter(i => i.cartId !== item.cartId)); else setCart(prev => prev.map(i => i.cartId === item.cartId ? {...i, quantity: i.quantity - 1} : i)); }} className="text-[#884466] hover:text-[#FF2A6D]"><Minus size={14} /></button>
                                  <span className="font-black text-[#FCE4EC] w-4 text-center text-xs">{item.quantity}</span>
                                  <button onClick={() => setCart(prev => prev.map(i => i.cartId === item.cartId ? {...i, quantity: i.quantity + 1} : i))} className="text-[#884466] hover:text-[#FF2A6D]"><Plus size={14} /></button>
                                </div>
                              </div>
                            ))
                          )}
                        </div>
                        {cart.length > 0 && (
                          <div className="p-6 border-t border-[#FF2A6D]/20 bg-[#0D0408]/80 space-y-6">
                            <div className="flex justify-between items-center text-xl font-black text-[#FCE4EC] italic tracking-tight">
                              <span className="text-[#AD6688] text-[9px] uppercase tracking-widest not-italic">TOTAL SACRIFICE</span>
                              <div className="flex flex-col items-end gap-0.5">
                                {hasVoucher && <span className="text-[#AD6688] text-xs line-through opacity-40">${(cartSubtotal + 2.99).toFixed(2)}</span>}
                                <span className="text-[#F48FB1] tabular-nums">{hasVoucher ? 'FREE' : `$${cartTotalDisplay.toFixed(2)}`}</span>
                              </div>
                            </div>
                            {hasVoucher ? (
                              <button disabled={isOrdering} onClick={handlePlaceOrder} className="w-full bg-gradient-to-r from-[#FF2A6D] via-[#9C27B0] to-[#880E4F] text-white py-5 rounded-2xl font-black text-lg tracking-widest italic active:scale-95 transition-all flex items-center justify-center gap-3 shadow-glow">{isOrdering ? <Loader2 className="animate-spin" size={20} /> : <><span className="mt-0.5 uppercase">Command Him</span> <Flame size={20} fill="currentColor" /></>}</button>
                            ) : (
                              <div className="space-y-4">
                                 <button onClick={() => setShowDevotionTest(true)} className="w-full bg-[#1A030C] border border-[#FF2A6D]/40 text-[#FF2A6D] py-4 rounded-xl font-black uppercase text-[9px] tracking-widest transition-all flex items-center justify-center gap-2"><Heart size={14} /> Beg Master (Free)</button>
                                 <div className="bg-[#0A0206]/60 p-4 rounded-2xl border border-[#330044] space-y-3">
                                   <p className="text-[#AD6688] text-[8px] uppercase tracking-widest font-black flex items-center gap-1 opacity-70"><CreditCard size={12}/> Pay with Plastic</p>
                                   <input type="text" placeholder="Card Number" value={ccDetails.number} onChange={e => setCcDetails({...ccDetails, number: e.target.value})} className="w-full bg-[#1A030C] border border-[#330044] px-4 py-2 rounded-lg text-xs outline-none text-white" />
                                   <div className="flex gap-2">
                                     <input type="text" placeholder="MM/YY" value={ccDetails.expiry} onChange={e => setCcDetails({...ccDetails, expiry: e.target.value})} className="w-1/2 bg-[#1A030C] border border-[#330044] px-4 py-2 rounded-lg text-xs outline-none text-white" />
                                     <input type="text" placeholder="CVC" value={ccDetails.cvc} onChange={e => setCcDetails({...ccDetails, cvc: e.target.value})} className="w-1/2 bg-[#1A030C] border border-[#330044] px-4 py-2 rounded-lg text-xs outline-none text-white" />
                                   </div>
                                   <button onClick={handleCCSubmit} disabled={isCcProcessing} className="w-full bg-[#330044]/50 text-[#AD6688] py-3 rounded-xl font-black uppercase text-[8px] tracking-widest flex items-center justify-center">{isCcProcessing ? <Loader2 className="animate-spin" size={12} /> : "Submit Currency"}</button>
                                 </div>
                              </div>
                            )}
                          </div>
                        )}
                      </div>
                    )}
                  </div>
                </div>
              )}

              <div className="fixed bottom-6 right-6 z-[100] flex flex-col items-end gap-3 sm:gap-4 font-black">
                <div className="flex gap-3 sm:gap-4 font-black">
                  <button onClick={() => { setIsInstaPortalOpen(!isInstaPortalOpen); setIsPortalOpen(false); }} className={`p-4 sm:p-6 rounded-full shadow-2xl transition-all border border-[#FF2A6D]/20 ${isInstaPortalOpen ? 'bg-[#FF2A6D] text-white shadow-glow' : 'bg-[#1A030C] text-[#AD6688]'}`}><Instagram size={22} className="sm:w-7 sm:h-7" /></button>
                  <button onClick={() => { setIsPortalOpen(!isPortalOpen); setIsInstaPortalOpen(false); }} className={`p-4 sm:p-6 rounded-full shadow-2xl transition-all border border-[#FF2A6D]/20 ${isPortalOpen ? 'bg-[#FF2A6D] text-white shadow-glow' : 'bg-[#1A030C] text-[#AD6688]'}`}>{isPortalOpen ? <X size={22} className="sm:w-7 sm:h-7" /> : <MessageCircle size={22} className="sm:w-7 sm:h-7" />}</button>
                </div>
              </div>
              <style>{`
                @import url('https://fonts.googleapis.com/css2?family=Playfair+Display:ital,wght@0,400;0,900;1,400;1,900&family=Montserrat:wght@400;900&display=swap');
                :root { font-family: 'Playfair Display', serif; }
                .no-scrollbar::-webkit-scrollbar { display: none; }
                .text-glow { text-shadow: 0 0 15px rgba(255, 42, 109, 0.6); }
                .shadow-glow { filter: drop-shadow(0 0 10px rgba(255, 42, 109, 0.4)); }
                .scrollbar-hide::-webkit-scrollbar { display: none; }
                .custom-scrollbar::-webkit-scrollbar { width: 3px; }
                .custom-scrollbar::-webkit-scrollbar-track { background: #0D0408; }
                .custom-scrollbar::-webkit-scrollbar-thumb { background: #FF2A6D33; border-radius: 10px; }
              `}</style>
            </div>
          );
        }

          return (
            <React.Fragment>
              <audio ref={audioRef} src="https://res.cloudinary.com/db8xboovt/video/upload/v1771934322/RINI_-_Red_Lights_ft._Wale_Official_Lyric_Video_odxpkl.mp3" loop preload="auto" autoPlay />
              {content}
            </React.Fragment>
          );
        };

        const root = createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
